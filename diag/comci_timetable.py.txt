def build_timetable_param(school_code: int, date_index: int = DEFAULT_DATE_INDEX) -> str:
    """시간표 API 용 Base64 쿼리 문자열 생성.

    원본 형식: ``73629_{school_code}_0_{date_index}`` → Base64 URL 쿼리.

    Args:
        school_code: 학교코드 (검색 API 에서 획득).
        date_index: 학생 시간표(/st) 일자 드롭다운 순서 (1=이번 주, 2=다음 주, …).
    """
    raw = f"{TIMETABLE_KEY}_{school_code}_{TIMETABLE_CACHE}_{date_index}"
    return base64.b64encode(raw.encode("utf-8")).decode("utf-8")

_DATE_SEP = re.compile(r"[-/.]")


def _parse_short_date(s: str) -> date:
    """``'26-03-16'`` 형태 → :class:`date` (2 자리 연도는 2000 대 가정)."""
    parts = _DATE_SEP.split(s.strip())
    if len(parts) != 3:
        raise ValueError(f"날짜 형식 아님: {s!r}")
    y, m, d = int(parts[0]), int(parts[1]), int(parts[2])
    if y < 100:
        y += 2000
    return date(y, m, d)


def _parse_week_range_label(label: str) -> Tuple[date, date]:
    """``'26-03-16 ~ 26-03-21'`` → ``(시작일, 종료일)``."""
    compact = re.sub(r"\s+", "", str(label))
    if "~" not in compact:
        raise ValueError(f"범위 구분 '~' 없음: {label!r}")
    left, _, right = compact.partition("~")
    return _parse_short_date(left), _parse_short_date(right)


def parse_ilja_weeks(ilja: Any) -> List[WeekInfo]:
    """API ``일자자료`` 배열을 :class:`WeekInfo` 리스트로 파싱.

    Returns:
        ``r`` 기준 오름차순 정렬된 주차 정보 리스트.
    """
    out: List[WeekInfo] = []
    if not isinstance(ilja, list):
        return out
    for row in ilja:
        if not isinstance(row, (list, tuple)) or len(row) < 2:
            continue
        try:
            r = int(row[0])
            start, end = _parse_week_range_label(str(row[1]))
        except (TypeError, ValueError):
            continue
        out.append(
            WeekInfo(r=r, label=str(row[1]).strip(), start=start, end=end)
        )
    return sorted(out, key=lambda w: w["r"])


def comci_weekday(d: date) -> Optional[Weekday]:
    """월=1 … 금=5. 주말이면 ``None``."""
    wd = d.weekday()  # 월=0
    if wd > 4:
        return None
    return wd + 1  # type: ignore[return-value]


def day_dates_for_week_range(start: date, end: date) -> Dict[int, date]:
    """해당 주 월~금(comci day 1..5)에 대응하는 실제 날짜 매핑.

    일자 범위 시작일이 월요일이 아니면, 그 날이 속한 ISO 주의 월요일을 기준으로 보정한다.
    """
    monday = start - timedelta(days=start.weekday())
    return {i: monday + timedelta(days=i - 1) for i in range(1, 6)}


def resolve_r_for_date(weeks: List[WeekInfo], on: date) -> Optional[int]:
    """달력 날짜가 속한 주의 드롭다운 인덱스 ``r``. 없으면 ``None``."""
    for w in weeks:
        if w["start"] <= on <= w["end"]:
            return w["r"]
    return None

#  컨텍스트 추출
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━


def extract_timetable_context(data: RawApiResponse, requested_r: int) -> TimetableContext:
    """원본 JSON 에서 시간표 메타 정보(주차·날짜 범위 등) 를 추출.

    ``requested_r``: 이번 요청에 사용한 ``date_index`` (URL 의 r 과 동일).
    """
    weeks = parse_ilja_weeks(data.get("일자자료"))

    today_r_raw = data.get("오늘r")
    today_r: Optional[int] = None
    if today_r_raw is not None:
        try:
            today_r = int(today_r_raw)
        except (TypeError, ValueError):
            pass

    # 활성 주 결정: 요청 r → 오늘 r → 첫 주
    selected: Optional[WeekInfo] = next(
        (w for w in weeks if w["r"] == requested_r), None
    )
    if selected is None and today_r is not None:
        selected = next((w for w in weeks if w["r"] == today_r), None)
    if selected is None and weeks:
        selected = weeks[0]

    day_dates: Dict[int, date] = {}
    if selected:
        day_dates = day_dates_for_week_range(selected["start"], selected["end"])

    return TimetableContext(
        requested_r=requested_r,
        today_r=today_r,
        weeks=weeks,
        일자자료=data.get("일자자료"),
        active_week=selected,
        day_dates=day_dates,
        학교명=data.get("학교명"),
        학년도=data.get("학년도"),
        시작일=data.get("시작일"),
    )

def _has_jaryo481(data: Dict[str, Any]) -> bool:
    """응답 JSON 이 ``자료481`` (또는 ``481`` 포함 키)을 갖는지 확인."""
    if "자료481" in data:
        return True
    return any("481" in str(k) for k in data)


def fetch_timetable_raw(
    school_code: int,
    date_index: int = DEFAULT_DATE_INDEX,
) -> RawApiResponse:
    """학교 시간표 원본 JSON 조회.

    Args:
        school_code: 학교코드 (검색 API 에서 획득).
        date_index: 주차 인덱스 ``r``.

    Returns:
        API 응답 JSON 전체 (``자료481`` 포함 객체 우선).

    Raises:
        requests.HTTPError: API 서버 오류.
    """
    param = build_timetable_param(school_code, date_index)
    url = f"{BASE_URL}?{param}"
    resp = fetch_json(url)
    return decode_json_matching(resp.text, _has_jaryo481)


def fetch_timetable_for_calendar_date(
    school_code: int,
    on: date,
) -> Tuple[int, RawApiResponse]:
    """특정 날짜가 포함된 주의 시간표 JSON 을 가져온다.

    ``일자자료`` / ``오늘r`` 로 r 을 고른 뒤, 필요하면 r 에 맞게 재요청한다.
    범위에 없으면 ``오늘r`` (없으면 1)로 조회한다.

    Returns:
        ``(r, raw_data)`` 튜플.
    """
    probe = fetch_timetable_raw(school_code, 1)
    weeks = parse_ilja_weeks(probe.get("일자자료"))
    r = resolve_r_for_date(weeks, on)

    if r is None:
        today_r_raw = probe.get("오늘r")
        try:
            r = int(today_r_raw) if today_r_raw is not None else DEFAULT_DATE_INDEX
        except (TypeError, ValueError):
            r = DEFAULT_DATE_INDEX

    if r == 1:
        return r, probe
    return r, fetch_timetable_raw(school_code, r)


def resolve_date_index_for_school(school_code: int, on: date) -> int:
    """``on`` 이 속한 주의 ``r`` 만 계산 (네트워크 1~2 회)."""
    r, _ = fetch_timetable_for_calendar_date(school_code, on)
    return r

#  자료 추출 헬퍼 (private)
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━


def _get_jaryo(data: RawApiResponse, num: str) -> NestedJaryoData:
    """``자료481`` 등 정식 키 우선, 없을 때만 키 스캔.

    Args:
        data: 원본 API JSON.
        num: 자료 번호 문자열 (``"481"``, ``"147"`` 등).

    Returns:
        해당 자료 배열. 찾지 못하면 빈 리스트.
    """
    canonical = f"자료{num}"
    if canonical in data:
        return data[canonical]
    # 키 이름이 미묘하게 다를 수 있음 – 번호 포함 + "자료" 포함 키 스캔
    for k in data:
        if num in str(k) and "자료" in str(k):
            return data[k]
    return []


def _safe_index(lst: Any, idx: int, default: Any = None) -> Any:
    """리스트 범위 안전 접근."""
    if isinstance(lst, list) and 0 <= idx < len(lst):
        return lst[idx]
    return default


def _get_code(
    data: NestedJaryoData,
    grade: int,
    class_num: int,
    day: int,
    period: int,
) -> int:
    """자료481/자료147 에서 코드 추출.

    실제 구조::

        [numGrades, grade0, grade1, ...]
        grade_i = [numClasses, class0, class1, ...]
        class_j = [5, day0, day1, ...]
        day_k   = [numPeriods, code1, code2, ...]

    Args:
        data: ``자료481`` 또는 ``자료147``.
        grade: 학년 (0-based 인덱스).
        class_num: 반 (0-based 인덱스).
        day: 요일 (1-based: 1=월).
        period: 교시 (1-based: 1=1교시).

    Returns:
        시간표 코드. 접근 불가 시 ``0``.
    """
    grade_data = _safe_index(data, grade + 1)
    if grade_data is None:
        return 0
    class_data = _safe_index(grade_data, class_num + 1)
    if class_data is None:
        return 0
    day_data = _safe_index(class_data, day)
    if not isinstance(day_data, list) or period >= len(day_data):
        return 0
    val = day_data[period]
    return int(val) if val else 0


def _get_room(
    data: NestedJaryoData,
    grade: int,
    class_num: int,
    day: int,
    period: int,
) -> str:
    """자료245 에서 강의실 문자열 추출 (자료481 과 동일 구조).

    Returns:
        강의실 이름. 접근 불가 시 빈 문자열.
    """
    grade_data = _safe_index(data, grade + 1)
    if grade_data is None:
        return ""
    class_data = _safe_index(grade_data, class_num + 1)
    if class_data is None:
        return ""
    day_data = _safe_index(class_data, day)
    if not isinstance(day_data, list) or period >= len(day_data):
        return ""
    val = day_data[period]
    return str(val) if val else ""

# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━


def _decode_code(
    code: int,
    split_val: int,
    subject_data: IndexArray,
    teacher_data: IndexArray,
) -> Tuple[str, str]:
    """시간표 코드를 과목·교사 이름으로 디코딩.

    ``split_val == 100`` 이면 ``teacher_idx = code // 100``, ``subject_idx = code % 100``.
    그 외에는 반대.

    Returns:
        ``(과목, 교사)`` 튜플.
    """
    if split_val == 100:
        teacher_idx = code // split_val
        subject_idx = code % split_val
    else:
        teacher_idx = code % split_val
        subject_idx = code // split_val

    subject = subject_data[subject_idx] if subject_idx < len(subject_data) else "미정"
    teacher = teacher_data[teacher_idx] if teacher_idx < len(teacher_data) else "미정"
    return subject, teacher


def _parse_room(raw: Any) -> str:
    """강의실 원시값에서 접두어(``idx_``) 를 제거."""
    s = str(raw) if raw else ""
    parts = s.split("_", 1)
    return parts[1] if len(parts) > 1 else parts[0]


def _parse_single_class(
    data: RawApiResponse,
    grade: int,
    class_num: int,
    day_dates: Optional[Dict[int, date]] = None,
    requested_r: Optional[int] = None,
) -> WeeklyTimetable:
    """특정 학년/반의 주간 시간표 파싱.

    Args:
        data: 원본 API JSON.
        grade: 학년 (0-based 인덱스).
        class_num: 반 (0-based 인덱스).
        day_dates: comci day(1~5) → 달력 날짜 매핑.
        requested_r: 현재 데이터의 ``r`` 값.

    Returns:
        ``{"월": [slot|None, ...], ..., "금": [...]}``
    """
    split_val: int = data.get("분리", 100)
    period_times: List[str] = data.get("일과시간", [])
    room_data = _get_jaryo(data, "245")
    base_data = _get_jaryo(data, "481")
    changed_data = _get_jaryo(data, "147")
    subject_data: IndexArray = _get_jaryo(data, "492")
    teacher_data: IndexArray = _get_jaryo(data, "446")

    timetable: WeeklyTimetable = {d: [] for d in DAYS_OF_WEEK}

    for period in range(1, MAX_PERIOD + 1):
        row: List[Optional[TimetableSlot]] = []
        for day in range(1, 6):  # 1=월 ~ 5=금
            base = _get_code(base_data, grade, class_num, day, period)
            chg = _get_code(changed_data, grade, class_num, day, period)
            code = chg if chg > 0 else base

            if not code or code <= EMPTY_CLASS_THRESHOLD:
                row.append(None)
                continue

            subject, teacher = _decode_code(
                code, split_val, subject_data, teacher_data
            )
            room = _parse_room(_get_room(room_data, grade, class_num, day, period))
            time_str = period_times[period - 1] if period <= len(period_times) else ""
            cal_date = day_dates.get(day) if day_dates else None

            row.append(
                TimetableSlot(
                    period=period,
                    time=time_str,
                    subject=subject,
                    teacher=teacher,
                    room=room,
                    changed=chg > 0 and chg != base,
                    r=requested_r,
                    day=day,
                    calendar_date=cal_date.isoformat() if cal_date else None,
                )
            )

        for idx, day_name in enumerate(DAYS_OF_WEEK):
            timetable[day_name].append(row[idx])

    return timetable

def parse_timetable(
    data: RawApiResponse,
    grade: Optional[int] = None,
    class_num: Optional[int] = None,
    *,
    requested_r: Optional[int] = None,
    enrich_calendar: bool = True,
    strict: bool = False,
    errors_out: Optional[List[str]] = None,
) -> Dict[str, WeeklyTimetable]:
    """시간표 JSON 파싱.

    Args:
        data: :func:`fetch_timetable_raw` 반환값.
        grade: 학년 (1-based). ``None`` 이면 전체.
        class_num: 반 (1-based). ``None`` 이면 전체.
        requested_r: 이번 데이터의 ``r``. 넣으면 슬롯에 ``r``·``calendar_date`` 채움.
        enrich_calendar: ``True`` 이고 ``requested_r`` 가 있으면 요일별 실제 날짜(ISO) 부착.
        strict: ``True`` 면 첫 파싱 예외를 그대로 전파.
        errors_out: 전체 반 순회 시 실패 항목 메시지를 ``append``.

    Returns:
        ``{"1학년 1반": WeeklyTimetable, ...}`` 형태.
        grade + class_num 둘 다 지정하면 해당 항목만 포함.
    """
    base_data = _get_jaryo(data, "481")
    if not base_data or not isinstance(base_data, list):
        return {}

    day_dates: Optional[Dict[int, date]] = None
    if requested_r is not None and enrich_calendar:
        ctx = extract_timetable_context(data, requested_r)
        day_dates = ctx.get("day_dates") or None

    num_grades = (
        int(base_data[0])
        if isinstance(base_data[0], (int, float))
        else len(base_data) - 1
    )

    result: Dict[str, WeeklyTimetable] = {}

    def _parse_one(g_idx: int, c_idx: int, g_disp: int, c_disp: int) -> None:
        key = f"{g_disp}학년 {c_disp}반"
        result[key] = _parse_single_class(
            data, g_idx, c_idx, day_dates=day_dates, requested_r=requested_r
        )

    # --- 특정 학년/반 ---
    if grade is not None and class_num is not None:
        g_idx, c_idx = grade - 1, class_num - 1
        if g_idx < num_grades and g_idx + 1 < len(base_data):
            grade_data = base_data[g_idx + 1]
            num_classes = (
                int(grade_data[0])
                if isinstance(grade_data, list) and grade_data
                else 0
            )
            if c_idx < num_classes:
                try:
                    _parse_one(g_idx, c_idx, grade, class_num)
                except (IndexError, KeyError, TypeError) as exc:
                    if strict:
                        raise
                    if errors_out is not None:
                        errors_out.append(
                            f"{grade}학년 {class_num}반: {type(exc).__name__}: {exc}"
                        )
        return result

    # --- 전체 ---
    for g_idx in range(num_grades):
        if g_idx + 1 >= len(base_data):
            continue
        grade_data = base_data[g_idx + 1]
        if not isinstance(grade_data, list):
            continue
        num_classes = (
            int(grade_data[0])
            if isinstance(grade_data[0], (int, float))
            else len(grade_data) - 1
        )
        for c_idx in range(num_classes):
            try:
                _parse_one(g_idx, c_idx, g_idx + 1, c_idx + 1)
            except (IndexError, KeyError, TypeError) as exc:
                if strict:
                    raise
                if errors_out is not None:
                    errors_out.append(
                        f"{g_idx + 1}학년 {c_idx + 1}반: {type(exc).__name__}: {exc}"
                    )

    return result


def timetable_for_calendar_day(
    weekly: WeeklyTimetable,
    on: date,
) -> DayPeriods:
    """파싱된 주간 시간표에서 특정 **달력** 하루의 교시 열만 반환.

    Args:
        weekly: ``{"월": [...], ..., "금": [...]}`` 형태.
        on: 조회할 날짜.

    Returns:
        해당 요일의 교시 리스트. 주말이면 빈 리스트.
    """
    d = comci_weekday(on)
    if d is None:
        return []
    return weekly.get(DAYS_OF_WEEK[d - 1], [])
