"""
``/th`` API 통합 클라이언트

:mod:`comci.client` 와 동일한 고수준 인터페이스(검색·달력 주차·컨텍스트)를
학급·교사 시간표 엔드포인트에 맞춘다.
"""

from __future__ import annotations

from datetime import date
from typing import Any, Dict, List, Optional, Union, overload

from ..school_search import search_schools
from ..timetable import extract_timetable_context, timetable_for_calendar_day
from ..types import (
    DayPeriods,
    DayTimetableWithContext,
    TimetableContext,
    TimetableWithContext,
    WeeklyTimetable,
)
from .timetable import (
    fetch_timetable_for_calendar_date,
    fetch_timetable_raw,
    parse_timetable,
    parse_teacher_timetable,
)


@overload
def get_timetable(
    school_code: int,
    grade: int,
    class_num: int,
    date_index: int = ...,
    on: Optional[date] = ...,
    *,
    with_context: bool = ...,
    enrich_calendar: bool = ...,
    strict_parse: bool = ...,
    parse_errors_out: Optional[List[str]] = ...,
) -> Union[WeeklyTimetable, TimetableWithContext]: ...


@overload
def get_timetable(
    school_code: int,
    grade: None = ...,
    class_num: None = ...,
    date_index: int = ...,
    on: Optional[date] = ...,
    *,
    with_context: bool = ...,
    enrich_calendar: bool = ...,
    strict_parse: bool = ...,
    parse_errors_out: Optional[List[str]] = ...,
) -> Union[Dict[str, WeeklyTimetable], TimetableWithContext]: ...


def get_timetable(
    school_code: int,
    grade: Optional[int] = None,
    class_num: Optional[int] = None,
    date_index: int = 1,
    on: Optional[date] = None,
    *,
    with_context: bool = False,
    enrich_calendar: bool = True,
    strict_parse: bool = False,
    parse_errors_out: Optional[List[str]] = None,
) -> Union[WeeklyTimetable, Dict[str, WeeklyTimetable], TimetableWithContext]:
    """``/th`` 학급별 시간표 조회. 파싱 규칙은 학생용과 동일."""
    if on is not None:
        used_r, raw = fetch_timetable_for_calendar_date(school_code, on)
    else:
        used_r = date_index
        raw = fetch_timetable_raw(school_code, used_r)

    context: TimetableContext = extract_timetable_context(raw, used_r)
    parsed = parse_timetable(
        raw,
        grade=grade,
        class_num=class_num,
        requested_r=used_r,
        enrich_calendar=enrich_calendar,
        strict=strict_parse,
        errors_out=parse_errors_out,
    )

    timetable: Union[WeeklyTimetable, Dict[str, WeeklyTimetable]]
    if grade is not None and class_num is not None and parsed:
        timetable = next(iter(parsed.values()))
    else:
        timetable = parsed

    if not with_context:
        return timetable
    return TimetableWithContext(context=context, timetable=timetable)


def get_day_timetable(
    school_code: int,
    on: date,
    grade: int,
    class_num: int,
    *,
    with_context: bool = False,
    enrich_calendar: bool = True,
) -> Union[DayPeriods, DayTimetableWithContext]:
    """특정 날짜 하루 교시 열만 반환. 주말이면 빈 리스트."""
    bundle = get_timetable(
        school_code,
        grade=grade,
        class_num=class_num,
        on=on,
        with_context=True,
        enrich_calendar=enrich_calendar,
    )
    assert isinstance(bundle, dict) and "timetable" in bundle
    day_rows = timetable_for_calendar_day(bundle["timetable"], on)  # type: ignore[arg-type]

    if not with_context:
        return day_rows
    return DayTimetableWithContext(context=bundle["context"], periods=day_rows)


def search_and_get_timetable(
    query: str,
    school_index: int = 0,
    grade: Optional[int] = None,
    class_num: Optional[int] = None,
    date_index: int = 1,
    on: Optional[date] = None,
    *,
    with_context: bool = False,
    enrich_calendar: bool = True,
) -> Union[WeeklyTimetable, Dict[str, WeeklyTimetable], TimetableWithContext]:
    """검색 → 학교 선택 → ``/th`` 시간표 조회."""
    schools = search_schools(query)
    if not schools:
        if with_context:
            return TimetableWithContext(
                context=TimetableContext(  # type: ignore[typeddict-item]
                    requested_r=date_index,
                    today_r=None,
                    weeks=[],
                    day_dates={},
                ),
                timetable={},
            )
        return {}

    school = schools[school_index]
    return get_timetable(
        school["school_code"],
        grade=grade,
        class_num=class_num,
        date_index=date_index,
        on=on,
        with_context=with_context,
        enrich_calendar=enrich_calendar,
    )


def get_teacher_timetable(
    school_code: int,
    date_index: int = 1,
    on: Optional[date] = None,
    *,
    grade: Optional[int] = None,
    class_num: Optional[int] = None,
    with_context: bool = False,
    enrich_calendar: bool = True,
    strict_parse: bool = False,
    parse_errors_out: Optional[List[str]] = None,
) -> Union[
    Dict[str, Dict[str, List[Optional[Dict[str, Any]]]]],
    Dict[str, Any],
]:
    """교사별 시간표. ``on`` 이 있으면 해당 주 ``r`` 로 조회.

    ``with_context=True`` 이면 ``{"context": TimetableContext, "timetable": …}`` 형태.
    """
    if on is not None:
        used_r, raw = fetch_timetable_for_calendar_date(school_code, on)
    else:
        used_r = date_index
        raw = fetch_timetable_raw(school_code, used_r)

    context: TimetableContext = extract_timetable_context(raw, used_r)
    tt = parse_teacher_timetable(
        raw,
        grade=grade,
        class_num=class_num,
        requested_r=used_r,
        enrich_calendar=enrich_calendar,
        strict=strict_parse,
        errors_out=parse_errors_out,
    )
    if not with_context:
        return tt
    return {"context": context, "timetable": tt}
